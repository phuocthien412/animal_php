    <script>
    document.addEventListener('DOMContentLoaded', function() {
        const token = localStorage.getItem('jwtToken');
        if (!token) {
            alert('Vui lòng đăng nhập');
            location.href = '/webbanhang/admin/login';
            return;
        }

        const productId = document.getElementById('product-id').value;
        const imagePreview = document.getElementById('image-preview');
        const existingImageInput = document.getElementById('existing_image');

        // Load product data
        fetch(`/webbanhang/api/product/${productId}`, {
            headers: {
                'Authorization': `Bearer ${token}`
            }
        })
        .then(response => response.json())
        .then(product => {
            document.getElementById('name').value = product.name;
            document.getElementById('description').value = product.description;
            document.getElementById('price').value = product.price;
            document.getElementById('category_id').value = product.category_id;
            existingImageInput.value = product.image;

            if (product.image) {
                imagePreview.src = '/webbanhang/' + product.image;
                imagePreview.style.display = 'block';
            }
        });

        // Load categories
        fetch('/webbanhang/api/category', {
            headers: {
                'Authorization': `Bearer ${token}`
            }
        })
        .then(response => response.json())
        .then(data => {
            const categorySelect = document.getElementById('category_id');
            data.categories.forEach(category => { // Thêm .categories để lấy array categories
                const option = document.createElement('option');
                option.value = category.id;
                option.textContent = category.name;
                categorySelect.appendChild(option);
            });
        });

        // Preview image before upload
        document.querySelector('input[name="image"]').addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    imagePreview.src = e.target.result;
                    imagePreview.style.display = 'block';
                }
                reader.readAsDataURL(file);
            }
        });
    });
    </script>
